import '../database/db_helper.dart';
import 'dart:math';

class SeedDataService {
  final DBHelper _db = DBHelper.instance;
  final Random _random = Random();

  Future<void> addSampleTransactions() async {
    final expenseCategories = [
      'Еда',
      'Транспорт',
      'Развлечения',
      'Продукты',
      'Кафе',
      'Такси',
      'Подписки',
      'Здоровье',
      'Одежда',
      'Другое',
    ];

    final merchants = {
      'Еда': ['Пятёрочка', 'Магнит', 'Перекрёсток', 'Лента'],
      'Транспорт': ['Метро', 'Автобус', 'Трамвай'],
      'Развлечения': ['Кино', 'Театр', 'Концерт'],
      'Продукты': ['Пятёрочка', 'Магнит', 'Дикси'],
      'Кафе': ['Starbucks', 'KFC', 'McDonalds', 'Кофемания'],
      'Такси': ['Яндекс.Такси', 'Uber'],
      'Подписки': ['Netflix', 'Spotify', 'Яндекс.Плюс'],
      'Здоровье': ['Аптека', 'Клиника'],
      'Одежда': ['Zara', 'H&M', 'LC Waikiki'],
      'Другое': ['Разное'],
    };

    // Получаем существующие счета
    final accounts = await _db.getAccounts();
    if (accounts.isEmpty) {
      print('❌ Нет счетов для добавления транзакций');
      return;
    }

    final accountId = accounts.first['id'];

    // Генерируем транзакции за последние 6 месяцев
    final now = DateTime.now();
    final startDate = DateTime(now.year, now.month - 6, 1);
    int transactionId = 1;

    for (var date = startDate; date.isBefore(now); date = date.add(const Duration(days: 1))) {
      // 1-3 транзакции в день (реалистично)
      final transactionsPerDay = _random.nextInt(3) + 1;

      for (var i = 0; i < transactionsPerDay; i++) {
        final category = expenseCategories[_random.nextInt(expenseCategories.length)];
        final merchantList = merchants[category]!;
        final merchant = merchantList[_random.nextInt(merchantList.length)];

        // 🔥 РЕАЛИСТИЧНЫЕ СУММЫ РАСХОДОВ
        double amount;

        switch (category) {
          case 'Продукты':
            // Продукты: 300-1500 ₽
            amount = 300 + _random.nextDouble() * 1200;
            break;
          case 'Еда':
            // Еда вне дома: 200-800 ₽
            amount = 200 + _random.nextDouble() * 600;
            break;
          case 'Кафе':
            // Кафе: 150-600 ₽
            amount = 150 + _random.nextDouble() * 450;
            break;
          case 'Транспорт':
            // Транспорт: 50-200 ₽
            amount = 50 + _random.nextDouble() * 150;
            break;
          case 'Такси':
            // Такси: 200-600 ₽
            amount = 200 + _random.nextDouble() * 400;
            break;
          case 'Развлечения':
            // Развлечения: 300-1500 ₽
            amount = 300 + _random.nextDouble() * 1200;
            break;
          case 'Подписки':
            // Подписки: 100-500 ₽ (ежемесячные)
            amount = 100 + _random.nextDouble() * 400;
            break;
          case 'Здоровье':
            // Здоровье: 200-2000 ₽
            amount = 200 + _random.nextDouble() * 1800;
            break;
          case 'Одежда':
            // Одежда: 1000-5000 ₽ (реже)
            amount = 1000 + _random.nextDouble() * 4000;
            break;
          default:
            // Другое: 100-1000 ₽
            amount = 100 + _random.nextDouble() * 900;
        }

        // Округляем до целых
        amount = amount.roundToDouble();

        await _db.insertTransaction({
          'title': merchant,
          'amount': amount,
          'type': 'expense',
          'category': category,
          'accountId': accountId,
          'date': date.add(Duration(hours: 8 + i * 3)).toString(),
        });

        transactionId++;
      }

      // 🔥 РЕАЛИСТИЧНЫЙ ДОХОД - зарплата 2 раза в месяц
      if (date.day == 5 || date.day == 20) {
        // Зарплата: 45000-50000 ₽ с небольшой вариацией
        final salary = 45000 + _random.nextInt(5000);

        await _db.insertTransaction({
          'title': 'Зарплата',
          'amount': salary.toDouble(),
          'type': 'income',
          'category': 'Зарплата',
          'accountId': accountId,
          'date': date.add(const Duration(hours: 10)).toString(),
        });

        transactionId++;
      }

      // 🔥 КРУПНЫЕ ПОКУПКИ - реже и реалистичнее
      // Раз в 2-3 месяца что-то крупное
      if (_random.nextInt(60) == 0) {
        final largeCategory = ['Одежда', 'Развлечения', 'Здоровье'][_random.nextInt(3)];
        double largeAmount;
        String title;

        if (largeCategory == 'Одежда') {
          largeAmount = 5000 + _random.nextDouble() * 10000; // 5000-15000 ₽
          title = 'Одежда';
        } else if (largeCategory == 'Развлечения') {
          largeAmount = 3000 + _random.nextDouble() * 7000; // 3000-10000 ₽
          title = 'Отдых/Путешествие';
        } else {
          largeAmount = 2000 + _random.nextDouble() * 5000; // 2000-7000 ₽
          title = 'Медицинские услуги';
        }

        await _db.insertTransaction({
          'title': title,
          'amount': largeAmount.roundToDouble(),
          'type': 'expense',
          'category': largeCategory,
          'accountId': accountId,
          'date': date.add(const Duration(hours: 14)).toString(),
        });

        transactionId++;
      }
    }

    print('✅ Добавлено тестовых транзакций: $transactionId');
  }

  Future<void> addSampleLimits() async {
    final limits = {
      'Еда': 8000.0,
      'Транспорт': 3000.0,
      'Развлечения': 5000.0,
      'Кафе': 3000.0,
      'Такси': 2000.0,
      'Подписки': 1000.0,
      'Продукты': 15000.0,
      'Здоровье': 5000.0,
      'Одежда': 10000.0,
    };

    final now = DateTime.now();
    for (var entry in limits.entries) {
      await _db.saveCategoryLimit(
        entry.key,
        entry.value,
        now.month,
        now.year,
      );
    }

    print('✅ Добавлено лимитов: ${limits.length}');
  }

  Future<void> seedAll() async {
    print('🌱 Начинаем добавление тестовых данных...');
    await addSampleTransactions();
    await addSampleLimits();
    print('✅ Тестовые данные добавлены успешно!');
  }
}
